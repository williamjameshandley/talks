                        \item AI builds tools and infrastructure

            \end{enumerate}

\begin{frame}
    \frametitle{Skills vs MCPs}
    \framesubtitle{Two ways to extend what the model knows}
    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \textbf{Skills (read-only)}
            \begin{itemize}
                \item \small{Text files loaded into context}
                \item \small{Best practices, physics, style guides}
                \item \small{Example: \texttt{cobaya.md}}
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{MCPs (read-write)}
            \begin{itemize}
                \item \small{Tools that execute actions}
                \item \small{Email, files, calendar}
                \item \small{Example: GTD task manager}
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{1em}
    \centering
    Knowledge $\to$ skill. Actions $\to$ MCP.\\
    \vspace{0.5em}
    \small{MCPs: \tthref{github.com/handley-lab/mcp-handley-lab}}\\
    \small{New: Word \& Excel MCPs, upgraded LLM interface}\\

\begin{frame}
    \frametitle{What is GTD?}
    \framesubtitle{David Allen's Getting Things Done}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item Core principle: get everything out of your head
                \item Five phases:
                    \begin{enumerate}
                        \item \textbf{Capture}: collect what has your attention
                        \item \textbf{Clarify}: what is it? is it actionable?
                        \item \textbf{Organise}: put it where it belongs
                        \item \textbf{Reflect}: review regularly
                        \item \textbf{Engage}: do the work
                    \end{enumerate}
                \item Frees your mind for thinking, not remembering
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{figures/gtd-what.jpg}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{GTD as external memory}
    \framesubtitle{Solving the stateless problem}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item Problem: LLMs forget when the context fills
                \item Solution: Store state externally
                    \begin{itemize}
                        \item Project: the goal
                        \item Next action: what to do now
                        \item Review: where you check progress
                    \end{itemize}
                \item Keeps a record of what was decided and why
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{figures/gtd-memory.jpg}
        \end{column}
    \end{columns}
\end{frame}
