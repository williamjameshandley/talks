\begin{frame}
    \frametitle{Two Revolutions}
    \framesubtitle{Distinct shifts, often conflated}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{\C[0]{GPUs for inference}}
            \begin{itemize}
                \item GPU hardware (developed for ML) accelerates \textit{classical} statistical methods
                \item Speedups of $100\times$--$5000\times$
                \item Rigorous Bayesian inference at previously impractical scales
            \end{itemize}
        \end{block}
        \column{0.48\textwidth}
        \begin{block}{\C[3]{LLMs for development}}
            \begin{itemize}
                \item Large Language Models change how we write code and manage research
                \item Not automating the science itself
                \item Reducing the technical overhead (verification is the new bottleneck)
            \end{itemize}
        \end{block}
    \end{columns}
    \begin{alertblock}{}
        GPUs make full Bayesian workflows feasible at scale. LLMs reduce the development time, but verification remains the limiting step.
    \end{alertblock}
\end{frame}

\begin{frame}
    \frametitle{Why Speed Matters}
    \framesubtitle{The scientific consequences of fast inference}
    \vspace{0.5em}
    \begin{itemize}
        \item \textbf{Speed}: If your likelihood takes 1s, you can do parameter estimation. If it takes 1ms, you can do \dots
        \item \textbf{Systematics}: Marginalise over 100+ nuisance parameters rather than fixing them.
        \item \textbf{Model Comparison}: Full Bayesian model comparison ($\mathcal{Z}$) between non-nested models.
        \item \textbf{Reproducibility}: If an analysis takes 10 minutes, everyone can rerun it. If it takes months, errors persist longer.
    \end{itemize}
\end{frame}

\begin{frame}

    \framesubtitle{Laplace approximation + auto-differentiation}
    \student{toby_lovick.jpg}{Toby Lovick}{PhD Student}
    \begin{columns}
        \column{0.49\textwidth}
        \begin{block}{Laplace Approximation + Optimisation}
            \begin{itemize}
                \item SNe have 100s of latent variables per object (beyond SALT2's 2).
                \item LAO: optimise to mode, auto-diff the Hessian, marginalise exactly.
                \item \textbf{25,602-D} likelihood in \textbf{11 minutes}.
            \end{itemize}
        \end{block}
        \begin{block}{Why This Matters}
            \begin{itemize}
                \item Nested Sampling at this scale: $D^3$ $\to$ a decade of H200 time.
                \item Planck needed manual per-parameter derivations. LAO just auto-diffs the forward model.
            \end{itemize}
        \end{block}

\begin{frame}
    \frametitle{Two implications}
    \framesubtitle{Rebound effects and maintenance burden}
    \begin{columns}
        \column{0.49\textwidth}
        \begin{block}{Jevons Paradox}
            \begin{itemize}
                \item In our experience LLMs don't reduce total time.
                \item They let you attempt larger analyses and iterate more.
                \item Tasks that used to require large teams become feasible for small projects --- provided verification is solid.
            \end{itemize}
        \end{block}
        \column{0.49\textwidth}
        \begin{block}{Code Debt}
            \begin{itemize}
                \item PhD students spend years wrestling with legacy Fortran monoliths.
                \item Long-term, the physics specification and tests should be primary, not inherited behaviour from old code.
                \item More modular implementations with strong regression tests.
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Trotta \& Hogg}
    \framesubtitle{Two serious arguments about AI in academia}
    \begin{columns}
        \column{0.49\textwidth}
        \begin{block}{Trotta \arxiv{2602.10165}}
            \begin{itemize}
                \item AI risks breaking the trust loop between student and understanding.
                \item If students generate code they don't understand, the verification chain fails.
            \end{itemize}
        \end{block}
        \begin{block}{Hogg \arxiv{2602.10181}}
            \begin{itemize}
                \item AI makes research more accessible and rigorous, not less.
                \item The bottleneck shifts to verification, which is a scientific skill.
            \end{itemize}
        \end{block}
        \column{0.49\textwidth}
        \begin{alertblock}{Our Experience}
            \begin{itemize}
                \item The dust law bug: found \textit{because} Sam's GPU rewrite forced rigorous validation.
                \item LLMs make wrong code easy. Testing makes it detectable.
                \item The scientific method applied to software.
            \end{itemize}
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Practical suggestions}
    \begin{enumerate}
        \item \textbf{Make likelihoods vectorisable}: Benchmark classical inference on GPUs before reaching for amortised Neural Networks.
        \item \textbf{Prioritise verification}: Regression tests and CI matter more now than raw coding speed.
        \item \textbf{Curate reference material}: Maintain version-controlled model/assumption notes and reference outputs that you can feed into tools.
    \end{enumerate}
\end{frame}
