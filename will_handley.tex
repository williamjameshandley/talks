\begin{frame}
    \frametitle{The three pillars of (Bayesian) inference}
    \vspace{-0.5em}
    \begin{columns}[t]
        \column{0.33\textwidth}
        \begin{block}{Parameter estimation}
            What do the data tell us about the parameters of a model?\\
            {\textit{e.g. the size or age of a $\Lambda$CDM universe}}%
            \[ \hspace{-4pt}\C[0]{P(\theta|D,M)} = \frac{\C[2]{P(D|\theta,M)} \C[1]{P(\theta|M)}}{\C[3]{P(D|M)}} \]
            \[ \C[0]{\mathcal{P}} = \frac{\C[2]{\mathcal{L}} \times\C[1]{\pi}}{\C[3]{\mathcal{Z}}}\]
            \[ \C[0]{\text{Posterior}} = \frac{\C[2]{\text{Likelihood}} \times\C[1]{\text{Prior}}}{\C[3]{\text{Evidence}}}\]
        \end{block}
        \column{0.3\textwidth}
        \begin{block}{Model comparison}
            How much does the data support a particular model?\\
            {\textit{e.g. $\Lambda$CDM vs a dynamic dark energy cosmology}}%
            \[ \C[5]{P(M|D)} = \frac{\C[3]{P(D|M)} \C[6]{P(M)}}{\C[7]{P(D)}} \vspace{-7pt}\]
            \[ \frac{\C[3]{\mathcal{Z}_{M}} \C[6]{\Pi_{M}}}{\C[7]{\sum_m Z_m \Pi_m}} \]
            \[ \C[5]{\text{Posterior}} = \frac{\C[3]{\text{Evidence}} \times\C[6]{\text{Prior}}}{\C[7]{\text{Normalisation}}}\]
        \end{block}
        \column{0.33\textwidth}
        \begin{block}{Tension quantification}
            Do different datasets make consistent predictions from the same model?
            {\textit{\textit{e.g. CMB vs Type IA supernovae data}}}%
            \[ \mathcal{R} = \frac{\C[3]{\mathcal{Z}}_{AB}}{\C[3]{\mathcal{Z}}_A\C[3]{\mathcal{Z}}_\mathcal{B}}\]
            \[
                \begin{aligned} \log\mathcal{S} = \av[{\C[0]{\mathcal{P}}_{AB}}]{\C[2]{\log\mathcal{L}}_{AB}}&\\
                    -\av[{\C[0]{\mathcal{P}}_{A}}]{\C[2]{\log\mathcal{L}}_{A}}&\\
                    -\av[{\C[0]{\mathcal{P}}_{B}}]{\C[2]{\log\mathcal{L}}_{B}}&
                \end{aligned}
            \]
        \end{block}
    \end{columns}
    \arxiv{1902.04029} \arxiv{1903.06682} \arxiv{1910.07820}

\begin{frame}
    \frametitle{Why $4.2\sigma$ doesn't settle it}
    \framesubtitle{The Jeffreys--Lindley paradox}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Frequentist ($\Delta\chi^2$)}
            \begin{itemize}
                \item Rejects the null ($w=-1$) when the best fit deviates \textit{at all}.
                \item With enough data, \textit{any} exact null is rejected --- $p \to 0$.
                \item $4.2\sigma$: ``the data are unlikely under $\Lambda$CDM.''
            \end{itemize}
        \end{block}
        \column{0.48\textwidth}
        \begin{exampleblock}{Bayesian (Bayes factor)}
            \begin{itemize}
                \item Compares full models: $\Lambda$CDM ($N$ params) vs $w_0 w_a$CDM ($N{+}2$).
                \item Extra parameters pay an Occam penalty --- prior volume wasted on bad regions.
                \item Unless the data \textit{concentrate} away from $w=-1$, $\Lambda$CDM wins.
            \end{itemize}
        \end{exampleblock}
    \end{columns}
    \vfill
    \begin{alertblock}{The paradox}
        As data grow, the two frameworks \alert{diverge}: arbitrarily significant $p$-values coexist with Bayes factors favouring the null. DESI's $3.2$--$4.2\sigma$ is exactly this regime.
    \end{alertblock}
\end{frame}

                \item \textit{``Write me a Python loop.''}
                \item Limited by user memory and syntax knowledge.
                \item Useful for snippets, bad for systems.
            \end{itemize}
        \end{block}
        \column{0.48\textwidth}
        \begin{block}{2026: Context engineering}
            \begin{itemize}
                \item \textit{``Here is the CAMB repo. Find the neutrino integration logic.''}
                \item LLMs acting on the \textbf{full context} of the project.
                \item The bottleneck is no longer generating code, but \keyterm{evaluating} it.
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}

    \frametitle{Conclusions}
    \vspace{-0.5em}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{\C[0]{GPUs $\neq$ AI}}
            \begin{itemize}
                \item Classical statistics on GPU is competitive with ML --- and \alert{interpretable}.
                \item GW PE: 2\,h $\to$ 30\,s. Dark energy: days $\to$ minutes. Supernovae: 50$\times$. The method transfers across astronomy.

        \end{block}

        \begin{exampleblock}{LLMs $\neq$ science}
            \begin{itemize}
                \item LLMs don't replace scientists --- they make code and text virtually free.

            \end{itemize}
        \end{exampleblock}
    \end{columns}
