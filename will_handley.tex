% --- Slide 1: The Inverse Problem ---
\begin{frame}
    \frametitle{The Inverse Problem}
    \framesubtitle{Forward and inverse problems}
    \begin{columns}[T]
        \column{0.49\textwidth}
        \begin{block}{The Forward Model}
            Parameters $\theta$ $\longrightarrow$ Observables $D$
            \begin{itemize}
                \item Orbital elements $\longrightarrow$ satellite positions
                \item Soil properties $\longrightarrow$ reflected signal
                \item The forward problem is usually straightforward
            \end{itemize}
        \end{block}
        \begin{block}{The Inverse Problem}
            Observables $D$ $\longrightarrow$ Parameters $\theta$
            \begin{itemize}
                \item Infer parameters from observables
                \item \keyterm{Ill-posed}: many $\theta$ can produce the same $D$
            \end{itemize}
        \end{block}
        \column{0.49\textwidth}
        \begin{block}{The Challenge}
            The parameter space is often:
            \begin{itemize}
                \item \keyterm{High-dimensional}: Many parameters to fit.
                \item \keyterm{Degenerate}: Different combinations of parameters can produce similar signals.
                \item \keyterm{Non-linear}: Small changes in $\theta$ can cause large changes in $D$.
            \end{itemize}
        \end{block}
        \begin{block}{This talk}
            Bayesian inference provides parameter constraints, uncertainties, and model comparison.
        \end{block}
    \end{columns}
\end{frame}

% --- Slide 6: From chi-squared to likelihood ---
\begin{frame}
    \frametitle{Probability distributions}
    \framesubtitle{Fitting lines to data}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{itemize}
            \item The residual $y_i - f(x_i;\theta)$ shown here
            \item If errors are Gaussian, the probability of observing a single datum:
                \[
                    P(y_i | \theta,M) = \frac{1}{\sqrt{2\pi}\sigma_i}\exp\!\left({-\frac{|y_i-f(x_i;\theta)|^2}{2\sigma_i^2}}\right)
                \]
            \item The probability of observing the dataset:
                \begin{align}
                    P(D | \theta,M) &\propto e^{-\chi^2(\theta)/2}
                    \nonumber
                \end{align}
        \end{itemize}
        \column{0.5\textwidth}
        \includegraphics[width=\textwidth]{figures/data_diff_1.pdf}
    \end{columns}

% --- Slide 7: Maximum likelihood ---
\begin{frame}
    \frametitle{Maximum likelihood}
    \framesubtitle{Fitting lines to data}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{itemize}
            \item Minimising $\chi^2(\theta)$ is equivalent to maximising $P(D|\theta,M) \propto e^{-\chi^2(\theta)/2}$
            \item ``Least squares'' $\equiv$ ``maximum likelihood'' \\(if data are Gaussian).
            \item However, one can go much further than least squares\ldots
        \end{itemize}
        \column{0.5\textwidth}
        \includegraphics[width=\textwidth]{figures/data_diff.pdf}
    \end{columns}
    \begin{block}{}
        Least squares is a special case of likelihood-based inference. This sits naturally within a Bayesian framework.
    \end{block}
\end{frame}

% --- Bayes' Theorem ---
\begin{frame}
    \frametitle{The Language of Inference}
    \framesubtitle{Maximum likelihood is only the beginning}
    \begin{columns}
        \column{0.33\textwidth}
            \begin{block}{\C[1]{Prior}\hfill $\C[1]{\pi(\theta)}$}
                What we believe about the parameters \textit{before} we see the data. Our physical constraints.
            \end{block}
        \vspace{2em}
        \begin{block}{\C[3]{Evidence}\hfill $\C[3]{\mathcal{Z}(D)}$}
            How well the model as a whole explains the data. The key to model comparison.
        \end{block}
        \column{0.33\textwidth}
\[\underbrace{\C[0]{\mathcal{P}(\theta|D)}}_{\C[0]{\text{Posterior}}} = \frac{\overbrace{\C[2]{\mathcal{L}(D|\theta)}}^{\C[2]{\text{Likelihood}}} \times \overbrace{\C[1]{\pi(\theta)}}^{\C[1]{\text{Prior}}}}{\underbrace{\C[3]{\mathcal{Z}(D)}}_{\C[3]{\text{Evidence}}}}\]
        \column{0.33\textwidth}
            \begin{block}{\C[2]{Likelihood}\hfill $\C[2]{\mathcal{L}(D|\theta)}$}
                Forward model + noise model. For Gaussian errors, $\C[2]{\mathcal{L}} \propto e^{-\chi^2/2}$.
            \end{block}
        \vspace{2em}
        \begin{block}{\C[0]{Posterior}\hfill $\C[0]{\mathcal{P}(\theta|D)}$}
            The solution to the inverse problem: parameters \textit{with error bars}.
        \end{block}
    \end{columns}
\end{frame}

% --- Slide 4: Why Sampling? ---
\begin{frame}
    \frametitle{Why do sampling?}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{itemize}
            \item The cornerstone of numerical Bayesian inference is working with \textbf{samples}.
            \item Generate a set of representative parameters drawn in proportion to the posterior $\theta\sim\mathcal{P}$.
            \item The magic of marginalisation $\Rightarrow$ perform usual analysis on each sample in turn.
            \item The golden rule is \textbf{stay in samples} until the last moment before computing summary statistics because \[\boxed{f(\:\av{X}\:)\ne \av{\:f(X)\:}}\]
            \item Generally need $\sim\mathcal{O}(12)$ independent samples to compute a value and error bar.
        \end{itemize}
        \column{0.5\textwidth}
        \vspace{-1.8em}
        \begin{center}
            \includegraphics[width=\textwidth]{figures/himmelblau_samples.pdf}
        \end{center}

\end{frame}

% --- Slide 5: MCMC ---
\begin{frame}
    \frametitle{Markov Chain Monte Carlo (MCMC)}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \only<1-5>{
        \begin{block}{How it Works (Metropolis-Hastings)}
            A ``walker'' exploring the parameter landscape.
            \begin{enumerate}
                \item Take a random step to a new position.
                \item If the new spot is ``higher'' (better fit), move there.
                \item If it's ``lower'', maybe move there anyway (with probability proportional to how much lower it is).
                \item Repeat millions of times. The path traces the posterior distribution.
            \end{enumerate}
        \end{block}
        }
        \only<6-9>{
            \vspace{-1em}
        \begin{block}{Advantages \& Limitations}
            \begin{itemize}
                \item Explores the full posterior and gives uncertainties.
                \item \textcolor{red}{Limitation:} The walker can get ``stuck'' in a local mode.
                \item \textcolor{red}{Limitation:} Slow to explore correlated parameters.
            \end{itemize}
        \end{block}
        \begin{alertblock}{}
            MCMC explores the posterior, but random-walk proposals can be inefficient in complex parameter spaces.
        \end{alertblock}
        }

% --- Slide 6: Evidence ---
\begin{frame}
    \frametitle{The Missing Piece: Why Evidence Calculation Matters}
    \framesubtitle{And why it's so hard to compute}
    \begin{columns}[T]
        \column{0.49\textwidth}
        \begin{block}{Why Evidence is Important}
            \begin{itemize}
                \item \textbf{Model Comparison}: Which model best explains the data?
            \[\mathcal{P}(M|D) \propto \mathcal{Z}(D|M) \mathcal{P}(M)\]
            e.g.\ Which drag model best explains the orbit? Is the reflection coherent or incoherent?
        \item  \textbf{Occam's Razor}: Automatic complexity penalty
            \[\log \mathcal{Z} = \langle \log \mathcal{L} \rangle_{\mathcal{P}} - \mathcal{D}_{\text{KL}}(\mathcal{P}||\pi)\]
            \end{itemize}
        \end{block}
        \column{0.49\textwidth}
        \begin{block}{Why Evidence is Hard}
            \begin{itemize}
                \item The high-dimensional evidence integral:
            \[\mathcal{Z} = \int \mathcal{L}(\theta) \pi(\theta) d\theta\]
            \pause
        \item The difficulty is not that most of parameter space has $\mathcal{L} \approx 0$...
            \pause
        \item The difficulty is that we can't estimate \textbf{volume} $d\theta$ in high dimensions!
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}

% --- Slide 7: Nested Sampling ---
\begin{frame}
    \frametitle{Nested Sampling}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \only<1-4>{
        \begin{block}{An alternative approach}
            Instead of a random walk, nested sampling proceeds from the outside in.
            \begin{enumerate}
                \item Start with ``live points'' scattered across the entire \keyterm{prior}.
                \item Find the point with the \textit{worst} likelihood and discard it.
                \item Replace it with a new point with a \textit{better} likelihood.
                \item The live points continuously ``shrink'' into regions of higher likelihood.
            \end{enumerate}
        \end{block}
        }
        \only<5-8>{
        \begin{block}{Key Advantages}
            \begin{itemize}
                \item Naturally handles \keyterm{multimodality}.
                \item Calculates the \keyterm{Bayesian Evidence} ($\C[3]{\mathcal{Z}}$) as a primary output.
                \item Gives posteriors \textit{and} evidence simultaneously.
            \end{itemize}
        \end{block}
        \begin{alertblock}{}
            Nested sampling is well suited to multimodal posteriors and directly computes the evidence.
        \end{alertblock}
        }
        \column{0.5\textwidth}
        \vspace{-1.8em}

                \item Nested sampling needs $\sim$1000 live points, each requiring many likelihood evaluations

        \begin{block}{For GNSS-R}
            \begin{itemize}
                \item Orbit determination: 6--15 parameters per satellite
                \item Constellation tracking: $\sim$10,000 satellites
                \item Real-time requirements
                \item This requires hardware acceleration
            \end{itemize}
        \end{block}
    \end{columns}
    \vspace{1em}
    \begin{alertblock}{}
        The limiting factor is computation. Practical deployment requires GPU acceleration.

\begin{frame}
    \frametitle{Ensemble Sampling (e.g.\ \texttt{emcee})}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \only<1-5>{
        \begin{block}{How it Works}
            Instead of one walker, use an \keyterm{ensemble} of hundreds.
            \begin{itemize}
                \item Walkers propose steps based on positions of \textit{other} walkers.
                \item The group learns the shape of the posterior.
                \item Much better at exploring correlations.
            \end{itemize}
        \end{block}
        }
        \only<6-9>{
        \vspace{-1em}
        \begin{block}{Advantages}
            \begin{itemize}
                \item Handles correlated parameters (e.g.\ roughness vs.\ dielectric constant).
                \item Easy to parallelise.
            \end{itemize}
        \end{block}
        \begin{block}{Limitation}
            Can still get trapped if modes are very far apart.
        \end{block}
        }

% --- Appendix: Volume Estimation ---
\begin{frame}
    \frametitle{How Nested Sampling Estimates Volumes: The Counting Trick}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{Volume Contraction}
            At each step, the volume contracts predictably:
            \[V_{i+1} = V_i \times \frac{n_{\text{inside}}}{n_{\text{total}}}\]
            indep.\ of dimensionality, geometry or topology
        \end{block}

        \begin{columns}[T]
            \column{0.45\textwidth}
            \begin{block}{Evidence}
                \[\mathcal{Z} = \sum \mathcal{L}_i \Delta V_i\]
            \end{block}

            \column{0.45\textwidth}
            \begin{block}{Posterior}
                \[w_i = \mathcal{L}_i \times \Delta V_i\]
            \end{block}
        \end{columns}

% --- Appendix: HMC ---
\begin{frame}
    \frametitle{Hamiltonian Monte Carlo (HMC)}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \only<1-5>{
        \begin{block}{How it Works}
            Uses gradients to guide exploration.
            \begin{enumerate}
                \item Treat parameters as ``particles'' with position and momentum.
                \item Use gradient of log-likelihood as ``force''.
                \item Propose coherent moves along gradient directions.
                \item Accept/reject using Metropolis criterion.
            \end{enumerate}
        \end{block}
        }
        \only<6-9>{
        \begin{block}{Advantages \& Requirements}
            \begin{itemize}
                \item Much more efficient than random walk for smooth posteriors.
                \item Requires gradients of the likelihood --- \texttt{jax.grad}!
                \item Can traverse parameter space much faster.
            \end{itemize}
        \end{block}
        \begin{alertblock}{Key Message}
            HMC leverages gradient information. With JAX, you get gradients for free.
        \end{alertblock}
        }

            \item \textbf{Complex, possibly multimodal?} $\rightarrow$ Nested Sampling.
            \item \textbf{Need to compare models?} $\rightarrow$ You \textit{must} use Nested Sampling.

    \end{block}
