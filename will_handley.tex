\begin{frame}
    \frametitle{Two Revolutions}
    \framesubtitle{Distinct shifts, often conflated}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{\C[0]{Revolution 1: GPUs for Science}}
            \begin{itemize}
                \item GPU hardware (developed for ML) accelerates \textit{classical} statistical methods
                \item Nested sampling speedups of 100--1000$\times$
                \item Rigorous Bayesian inference at previously impractical scales
            \end{itemize}
        \end{block}
        \column{0.48\textwidth}
        \begin{block}{\C[3]{Revolution 2: AI for Scientists}}
            \begin{itemize}
                \item LLMs change how we write code and manage research
                \item Not automating the science itself
                \item Reducing the technical overhead that consumes researcher time
            \end{itemize}
        \end{block}
    \end{columns}
    \begin{alertblock}{The Argument}
        These are distinct. One accelerates the \textit{calculation}; the other accelerates the \textit{researcher}. They may be more separable than current discourse implies.
    \end{alertblock}
\end{frame}

\begin{frame}
    \frametitle{Why do sampling?}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{itemize}
            \item The cornerstone of numerical Bayesian inference is working with \textbf{samples}.
            \item Generate a set of representative parameters drawn in proportion to the posterior $\theta\sim\mathcal{P}$.
            \item The magic of marginalisation $\Rightarrow$ perform usual analysis on each sample in turn.
            \item The golden rule is \textbf{stay in samples} until the last moment before computing summary statistics/triangle plots because \[\boxed{f(\:\av{X}\:)\ne \av{\:f(X)\:}}\]
            \item Generally need $\sim\mathcal{O}(12)$ independent samples to compute a value and error bar.
        \end{itemize}
        \column{0.5\textwidth}
        \vspace{-1.8em}
        \begin{center}
            \includegraphics[width=\textwidth]{figures/himmelblau_samples.pdf}
        \end{center}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Sampling Methods: A Brief Tour}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{MCMC (e.g. \texttt{emcee})}
            \begin{itemize}
                \item Random walk explores posterior
                \item Good for simple, well-behaved posteriors
                \item Can get stuck in local modes
                \item Posterior samples only
            \end{itemize}
        \end{block}
        \begin{block}{HMC/NUTS (e.g. \texttt{stan}, \texttt{numpyro})}
            \begin{itemize}
                \item Uses gradients to guide exploration
                \item Much more efficient for smooth posteriors
                \item Requires differentiable likelihoods
                \item Posterior samples only
            \end{itemize}
        \end{block}
        \column{0.48\textwidth}
        \begin{block}{Nested Sampling (e.g. \texttt{dynesty})}
            \begin{itemize}
                \item Explores from outside-in
                \item Naturally handles multimodality
                \item \textcolor{green!50!black}{Computes Bayesian Evidence}
                \item Essential for model comparison (i.e. science)
            \end{itemize}
        \end{block}
        \begin{alertblock}{}
            \textit{``The evidence $\mathcal{Z}$ is often the single most important number in the problem and I think every effort should be devoted to calculating it.''} \hfill --- MacKay (2003)
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Nested Sampling}
    \framesubtitle{Exploring from the outside-in}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \vspace{-10pt}
        \only<1-8>{
        \begin{block}{A Radically Different Approach}
            Instead of random walking, nested sampling attacks the problem from the outside-in.
            \begin{enumerate}
                \item Start with a set of ``live points'' scattered across the entire \keyterm{prior}.
                \item At each step: find the point with the \textit{worst} likelihood and discard it.
                \item Replace it with a new point drawn from the prior, but with a likelihood \textit{better} than the point you just discarded.
                \item This forces the set of live points to continuously ``shrink'' into regions of higher and higher likelihood.
            \end{enumerate}
        \end{block}
        }
        \only<9>{
        \begin{block}{Why This Works}
            \begin{itemize}
                \item Naturally handles \keyterm{multimodality}: the shrinking cloud finds all modes simultaneously.
                \item Estimates \keyterm{volume} probabilistically:
                    \[\frac{\C[1]{V_\mathrm{after}}}{\C[0]{V_\mathrm{before}}} \approx \frac{\C[1]{n_\mathrm{in}}}{\C[0]{n_\mathrm{out}}+\C[1]{n_\mathrm{in}}}\]
                \item Computes \C[3]{evidence $\mathcal{Z}$} and \keyterm{information} $\mathcal{D}_\mathrm{KL}$:
                    \[\C[3]{\log\mathcal{Z}} = \av[{\C[0]{\mathcal{P}}}]{\C[2]{\log\mathcal{L}}} - \mathcal{D}_\text{KL}\]
            \end{itemize}
        \end{block}
        }
        \column{0.49\textwidth}
        \vspace{-1.8em}
        \begin{center}
            \includegraphics<1>[width=\textwidth,page=1]{figures/himmelblau_ns.pdf}%
            \includegraphics<2>[width=\textwidth,page=2]{figures/himmelblau_ns.pdf}%

\begin{frame}
    \frametitle{The Computational Bottleneck}
    \framesubtitle{Why the future of astronomical inference is parallel computing}
    \begin{columns}
        \column{0.42\textwidth}
        \begin{block}{The Problem}
            \begin{itemize}
                \item Nested sampling is powerful but \textbf{slow}
                \item Complex parameter spaces require many likelihood evaluations
                \item Traditional CPU approaches struggle to scale
            \end{itemize}
        \end{block}

        \begin{center}
            \includegraphics[width=0.9\textwidth]{figures/gpu_vs_cpu.jpg}
        \end{center}

        \column{0.54\textwidth}
        \vspace{-1em}
        \begin{block}{CPU vs GPU: Different Approaches}
            \begin{itemize}
                \item \textbf{CPU}: Few powerful cores ($\sim$10s), sequential tasks
                \item \textbf{GPU}: Many simple cores ($\sim$1000s), parallel tasks
            \end{itemize}
        \end{block}
        \vspace{-0.5em}

        \begin{block}{Which Algorithms Can Exploit This?}
            \begin{itemize}
                \item Traditional MCMC/HMC: single walker $\to$ \textcolor{red}{no}
                \item \texttt{emcee}: ensemble of walkers $\to$ \textcolor{green!50!black}{yes}
                \item Nested sampling: many live points $\to$ \textcolor{green!50!black}{yes}
            \end{itemize}
        \end{block}
        \vspace{-0.5em}

        \begin{alertblock}{The Future is GPU}
            All new HPC facilities are GPU-based. We must adapt our codes.
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Two Independent Capabilities}
    \framesubtitle{A common misconception in scientific computing}
    \begin{center}
        \textbf{Differentiable programming languages}: JAX, PyTorch, TensorFlow, Julia, Stan, \ldots
    \end{center}
    \vspace{-5pt}
    \begin{columns}
        \column{0.48\textwidth}
        \begin{block}{Capability 1: Free Gradients}
            \begin{itemize}
                \item \textbf{Automatic differentiation}: $\nabla_\theta \log \mathcal{L}(\theta)$.
                \item Enables gradient-based MCMC (HMC, NUTS).
                \item Essential for modern optimization.
            \end{itemize}
        \end{block}
        \begin{block}{Traditional Physics Benefits}
            \begin{itemize}
                \item \textbf{Nested sampling}: Massive parallelization.
                \item \textbf{21cm signals}: Vectorized across frequency/time/angle.
                \item \textbf{N-body sims}: GPU acceleration.
            \end{itemize}
        \end{block}
        \column{0.48\textwidth}
        \begin{block}{Capability 2: GPU Parallelization}
            \begin{itemize}
                \item \textbf{Vectorization across ensembles}.
                \item Run 1000s of parallel chains/particles.
                \item Evaluate likelihoods simultaneously.
            \end{itemize}
        \end{block}
        \begin{block}{Key Insight: Often Confused}
            \begin{center}
                \textbf{These are completely independent.}\\
                \textbf{People mistake one for the other.}\\
                You can use gradients on CPU.\\
                You can GPU parallelize without gradients.\\
                \textbf{They serve different purposes.}
            \end{center}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{The Second Revolution: AI for Scientists}
    \framesubtitle{Shifting gears: How did we build this code?}
    \begin{columns}[c]
        \column{0.55\textwidth}
        \begin{itemize}
            \item Writing high-performance GPU code (CUDA, JAX) is historically difficult
            \item The ``80/20'' rule of scientific work:
                \begin{itemize}
                    \item 20\% ``hard thinking'' (physics, algorithms)
                    \item 80\% ``boring tasks'' (boilerplate, plotting, formatting)
                \end{itemize}
            \item LLMs are not about ``automating science'' (the 20\%)
            \item They collapse the 80\% overhead, letting us focus on what matters
            \item BlackJAX NS: core algorithm hand-coded; matplotlib, tests, docs vibe-coded
        \end{itemize}

\begin{frame}
    \frametitle{Three Ways to Use AI in Research}
    \begin{columns}[c]
        \column{0.55\textwidth}
        \begin{enumerate}
            \item \textbf{Foundation models}: Ask questions, get answers
                \begin{itemize}
                    \item Good for brainstorming
                    \item Limited for complex tasks
                \end{itemize}
            \item \textbf{Autonomous agents}: Let it run unsupervised
                \begin{itemize}
                    \item Generates volume, not quality
                    \item No accountability
                \end{itemize}
            \item \textbf{Supervised execution}: Human directs, AI executes
                \begin{itemize}
                    \item You provide intent and verify output
                    \item AI builds tools and infrastructure
                \end{itemize}
        \end{enumerate}
        \begin{alertblock}{The Sweet Spot}
            Supervised execution: Human provides \textbf{intent} + \textbf{verification}. AI provides \textbf{implementation}.
        \end{alertblock}

\begin{frame}
    \frametitle{The Simplest Approach: Optimization (e.g., $\chi^2$ Minimization)}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \only<1-6>{
        \begin{block}{How it Works: Hill Climbing}
            Imagine the parameter space is a landscape where lower $\chi^2$ (or higher likelihood) is ``downhill''.
            \begin{itemize}
                \item Start somewhere.
                \item Follow the steepest gradient downhill.
                \item Stop when you reach the bottom of a valley.
            \end{itemize}
        \end{block}
        \begin{block}{Advantages}
            \begin{itemize}
                \item \keyterm{Fast} and computationally cheap.
                \item Good for a quick first look.
            \end{itemize}
        \end{block}
        }
        \only<7-10>{
        \begin{block}{Limitations}
            \begin{itemize}
                \item Only gives a single \keyterm{point estimate} (the ``best fit'').
                \item \textbf{No uncertainty quantification!} Where are the error bars?
                \item Can easily get stuck in a \keyterm{local minimum}, missing the true global best fit.
            \end{itemize}
        \end{block}
        \begin{alertblock}{Key Message}
            Optimization is fast but gives an incomplete and potentially misleading picture. Science needs error bars.
        \end{alertblock}
        }

\begin{frame}
    \frametitle{The Classic Workhorse: Markov Chain Monte Carlo (MCMC)}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \only<1-5>{
        \begin{block}{How it Works (Metropolis-Hastings)}
            Imagine a ``walker'' exploring the parameter landscape.
            \begin{enumerate}
                \item Take a random step to a new position.
                \item If the new spot is ``higher'' (better likelihood), move there.
                \item If it's ``lower'', maybe move there anyway (with probability proportional to how much lower it is).
                \item Repeat millions of times. The path the walker takes traces the posterior distribution.
            \end{enumerate}
        \end{block}
        }
        \only<6-9>{
            \vspace{-1em}
        \begin{block}{Advantages \& Limitations}
            \begin{itemize}
                \item Explores the full posterior and gives uncertainties.
                \item \textcolor{red}{Limitation:} The walker can be inefficient. It can get ``stuck'' in a local high-likelihood region and fail to find other, separate modes.
                \item \textcolor{red}{Limitation:} Can be slow to explore highly correlated (``banana-shaped'') posteriors.
            \end{itemize}
        \end{block}
        \begin{alertblock}{Key Message}
            MCMC is a foundational sampling method, but its simple ``random walk'' can be inefficient in the complex parameter spaces of astronomical inference.
        \end{alertblock}
        }

\begin{frame}
    \frametitle{Gradient-Guided Sampling: Hamiltonian Monte Carlo (HMC)}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \only<1-5>{
        \begin{block}{How it Works}
            Uses gradients to guide exploration more efficiently than random walks.
            \begin{enumerate}
                \item Treat parameters as ``particles'' with position and momentum.
                \item Use gradient of log-likelihood as ``force'' to guide movement.
                \item Propose coherent moves along gradient directions.
                \item Accept/reject using Metropolis criterion.
            \end{enumerate}
        \end{block}
        }
        \only<6-9>{
        \begin{block}{Advantages \& Requirements}
            \begin{itemize}
                \item Much more efficient than random walk for smooth posteriors.
                \item Requires gradients of the likelihood function.
                \item Can traverse parameter space much faster.
                \item Less likely to get stuck in local regions.
            \end{itemize}
        \end{block}
        \begin{alertblock}{Key Message}
            HMC leverages gradient information for efficient sampling, but requires differentiable models.
        \end{alertblock}
        }
