\begin{frame}
    \frametitle{Where does the speed come from?}
    \framesubtitle{Example 1: Real-time gravitational wave parameter estimation}
    \student{metha_prathaban.jpg}{Metha Prathaban}{PhD Student}
    \begin{columns}
        \column{0.54\textwidth}
        \begin{block}{The Bottleneck}
            \begin{itemize}
                \item \textbf{Intuition}: Waveform generation (physics) is slow.
                \item \textbf{Reality}: Memory bandwidth for extrinsic parameters is the GPU bottleneck.
                \item \textbf{Key}: Dual-axis parallelisation --- over sampling particles \textit{and} frequencies in the likelihood. Opens a different scaling relation for next-gen detectors.
            \end{itemize}
        \end{block}
        \column{0.44\textwidth}
        \centering
        {\small \textbf{Inter-likelihood}: parallelise over particles}\\[0.3em]
        \begin{tikzpicture}[scale=0.9,
            dot/.style={C0, fill=C0}
        ]
            \def\numdots{50} \def\walksteps{6}
            \def\numoutermost{20}
            \def\xmax{3.75} \def\ymax{3.75}
            \def\cx{2} \def\cy{2}
            \pgfmathsetseed{\number\pdfrandomseed}
            \foreach \i in {1,...,\numdots} {
                \pgfmathsetmacro{\randx}{2 + (\xmax - 2) * rand}
                \pgfmathsetmacro{\randy}{2 + (\ymax - 2) * rand}
                \fill[dot] (\randx, \randy) circle (1.5pt);
                \pgfmathsetmacro{\distsq}{((\randx-\cx)*(\randx-\cx) + (\randy-\cy)*(\randy-\cy))}
                \expandafter\xdef\csname qx\i\endcsname{\randx}
                \expandafter\xdef\csname qy\i\endcsname{\randy}
                \expandafter\xdef\csname distsq\i\endcsname{\distsq}
            }
            \foreach \j in {1,...,\numoutermost} {
                \xdef\currentmaxdistsq{-1} \xdef\currentmaxidx{0}
                \foreach \i in {1,...,\numdots} {
                    \ifdim\csname distsq\i\endcsname pt > \currentmaxdistsq pt
                        \xdef\currentmaxdistsq{\csname distsq\i\endcsname}
                        \xdef\currentmaxidx{\i}
                    \fi
                }
                \expandafter\xdef\csname finaloutermost\j\endcsname{\currentmaxidx}
                \expandafter\xdef\csname distsq\currentmaxidx\endcsname{-1}
            }
            \foreach \j in {1,...,\numoutermost} {
                \pgfmathsetmacro{\startindex}{int(\csname finaloutermost\j\endcsname)}
                \edef\coordx{\csname qx\startindex\endcsname}
                \edef\coordy{\csname qy\startindex\endcsname}
                \draw[C3, thin] (\coordx, \coordy) circle (5pt);
            }
            \def\stepsize{0.6}
            \foreach \j in {1,...,\numoutermost} {
                \pgfmathsetmacro{\startindex}{int(\csname finaloutermost\j\endcsname)}
                \xdef\currentx{\csname qx\startindex\endcsname}
                \xdef\currenty{\csname qy\startindex\endcsname}
                \foreach \i in {1,...,\walksteps} {
                    \pgfmathsetmacro{\nextxpotential}{\currentx + \stepsize * cos(random(0,360))}
                    \pgfmathsetmacro{\nextypotential}{\currenty + \stepsize * sin(random(0,360))}
                    \pgfmathsetmacro{\nextx}{max(0.1, min(\xmax - 0.2, \nextxpotential))}
                    \pgfmathsetmacro{\nexty}{max(0.1, min(\ymax - 0.2, \nextypotential))}
                    \draw[C3] (\currentx, \currenty) -- (\nextx, \nexty);
                    \xdef\currentx{\nextx}
                    \xdef\currenty{\nexty}
                }
                \fill[C3!40!white] (\currentx, \currenty) circle (2pt);
            }
        \end{tikzpicture}
        \\[0.5em]
        {\small \textbf{Intra-likelihood}: parallelise over frequencies}\\[0.3em]
        \begin{tikzpicture}[scale=0.9,
            dot/.style={fill=black}
        ]
            \def\waveformplot{plot[domain=0.5:5.2, samples=200, smooth] (\x, {0.2 * \x^0.8 * sin(\x*\x*100)})}
            \draw[C0, line width=0.8pt] \waveformplot;
            \foreach \xpos in {0.6, 0.8, 1, 1.2, 1.4, 1.6, 1.8, 2, 2.2, 2.4, 2.6, 2.8, 3, 3.2, 3.4, 3.6, 3.8, 4.0, 4.2, 4.4, 4.6, 4.8, 5, 5.2} {
                \pgfmathsetmacro{\ypos}{0.2 * \xpos^0.8 * sin(\xpos*\xpos*100)}
                \fill[dot] (\xpos, \ypos) circle (1.2pt);
            }
        \end{tikzpicture}

    \end{columns}
    \vfill
    \hfill{\tiny \github{mrosep/talks}}

\begin{frame}
    \frametitle{Two implications}
    \framesubtitle{Rebound effects and maintenance burden}
    \begin{columns}
        \column{0.49\textwidth}
        \begin{block}{Jevons Paradox}
            \begin{itemize}
                \item In our experience LLMs don't reduce total time.
                \item They let you attempt larger analyses and iterate more.
                \item Tasks that used to require large teams become feasible for small projects --- provided verification is solid.
            \end{itemize}
        \end{block}
        \column{0.49\textwidth}
        \begin{block}{Code Debt}
            \begin{itemize}
                \item PhD students spend years wrestling with legacy Fortran monoliths.
                \item Long-term, the physics specification and tests should be primary, not inherited behaviour from old code.
                \item More modular implementations with strong regression tests.
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Trotta \& Hogg}
    \framesubtitle{Two serious arguments about AI in academia}
    \begin{columns}
        \column{0.49\textwidth}
        \begin{block}{Trotta \arxiv{2602.10165}}
            \begin{itemize}
                \item AI risks breaking the trust loop between student and understanding.
                \item If students generate code they don't understand, the verification chain fails.
            \end{itemize}
        \end{block}
        \begin{block}{Hogg \arxiv{2602.10181}}
            \begin{itemize}
                \item AI makes research more accessible and rigorous, not less.
                \item The bottleneck shifts to verification, which is a scientific skill.
            \end{itemize}
        \end{block}
        \column{0.49\textwidth}
        \begin{alertblock}{Our Experience}
            \begin{itemize}
                \item The dust law bug: found \textit{because} Sam's GPU rewrite forced rigorous validation.
                \item LLMs make wrong code easy. Testing makes it detectable.
                \item The scientific method applied to software.
            \end{itemize}
        \end{alertblock}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Practical suggestions}
    \begin{enumerate}
        \item \textbf{Make likelihoods vectorisable}: Benchmark classical inference on GPUs before reaching for amortised Neural Networks.
        \item \textbf{Prioritise verification}: Regression tests and CI matter more now than raw coding speed.
        \item \textbf{Curate reference material}: Maintain version-controlled model/assumption notes and reference outputs that you can feed into tools.
    \end{enumerate}
\end{frame}
